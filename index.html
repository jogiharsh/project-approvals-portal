-- 1. Enable Row-Level Security on app_state to block direct API table queries ALTER TABLE app_state ENABLE ROW LEVEL SECURITY; -- 2. Restrict direct API table access for anonymous users DROP POLICY IF EXISTS "deny_all_anon" ON app_state; CREATE POLICY "deny_all_anon" ON app_state FOR ALL TO anon USING (false) WITH CHECK (false); -- 3. Secure RPC: Fetch single client project (Filters out sensitive fields) CREATE OR REPLACE FUNCTION get_client_project(p_token TEXT) RETURNS jsonb LANGUAGE plpgsql SECURITY DEFINER SET search_path = public, pg_temp AS $$ DECLARE v_elem jsonb; BEGIN SELECT elem INTO v_elem FROM app_state, jsonb_array_elements(projects) AS elem WHERE elem->>'clientToken' = p_token LIMIT 1; IF v_elem IS NULL THEN RETURN NULL; END IF; -- Return ONLY non-sensitive client portal fields RETURN jsonb_build_object( 'id', v_elem->'id', 'siteAddress', v_elem->'siteAddress', 'clientName', v_elem->'clientName', 'approvalType', v_elem->'approvalType', 'council', v_elem->'council', 'clientToken', v_elem->'clientToken', 'documents', COALESCE(( SELECT jsonb_agg(jsonb_build_object( 'id', d->'id', 'name', d->'name', 'required', d->'required', 'status', d->'status', 'fileUrl', d->'fileUrl' )) FROM jsonb_array_elements(COALESCE(v_elem->'documents','[]'::jsonb)) AS d ), '[]'::jsonb), 'cloudFiles', COALESCE(v_elem->'cloudFiles','[]'::jsonb) ); END; $$; REVOKE ALL ON FUNCTION get_client_project(TEXT) FROM PUBLIC; GRANT EXECUTE ON FUNCTION get_client_project(TEXT) TO anon; -- 4. Secure RPC: Atomic client file attachment CREATE OR REPLACE FUNCTION attach_client_file( p_token TEXT, p_doc_id TEXT, p_file_path TEXT, p_file_url TEXT, p_file_name TEXT ) RETURNS void LANGUAGE plpgsql SECURITY DEFINER SET search_path = public, pg_temp AS $$ DECLARE v_idx INT; BEGIN SELECT (ord - 1) INTO v_idx FROM app_state, jsonb_array_elements(projects) WITH ORDINALITY AS e(elem, ord) WHERE elem->>'clientToken' = p_token LIMIT 1; IF v_idx IS NULL THEN RAISE EXCEPTION 'invalid_token'; END IF; UPDATE app_state SET projects = jsonb_set( jsonb_set( projects, ARRAY[v_idx::text, 'cloudFiles'], COALESCE(projects->v_idx->'cloudFiles','[]'::jsonb) || jsonb_build_array(jsonb_build_object( 'id', gen_random_uuid()::text, 'name', p_file_name, 'path', p_file_path, 'url', p_file_url )) ), ARRAY[v_idx::text, 'documents'], ( SELECT jsonb_agg( CASE WHEN d->>'id' = p_doc_id THEN jsonb_set(jsonb_set(d, '{status}', '"Received"'::jsonb), '{fileUrl}', to_jsonb(p_file_url)) ELSE d END ) FROM jsonb_array_elements(projects->v_idx->'documents') AS d ) ), updated_at = now() WHERE id = 'main'; RETURN; END; $$; REVOKE ALL ON FUNCTION attach_client_file(TEXT,TEXT,TEXT,TEXT,TEXT) FROM PUBLIC; GRANT EXECUTE ON FUNCTION attach_client_file(TEXT,TEXT,TEXT,TEXT,TEXT) TO anon; -- 5. Admin Read/Write RPC Helpers CREATE OR REPLACE FUNCTION admin_get_app_state() RETURNS jsonb LANGUAGE plpgsql SECURITY DEFINER SET search_path = public, pg_temp AS $$ DECLARE v_result jsonb; BEGIN SELECT row_to_json(a)::jsonb INTO v_result FROM app_state a WHERE id = 'main'; RETURN v_result; END; $$; CREATE OR REPLACE FUNCTION admin_save_app_state(p_payload jsonb) RETURNS void LANGUAGE plpgsql SECURITY DEFINER SET search_path = public, pg_temp AS $$ BEGIN INSERT INTO app_state (id, projects, "checklistLibrary", updated_at) VALUES ( 'main', COALESCE(p_payload->'projects', '[]'::jsonb), COALESCE(p_payload->'checklistLibrary', '{}'::jsonb), now() ) ON CONFLICT (id) DO UPDATE SET projects = EXCLUDED.projects, "checklistLibrary" = EXCLUDED."checklistLibrary", updated_at = now(); END; $$; GRANT EXECUTE ON FUNCTION admin_get_app_state() TO anon; GRANT EXECUTE ON FUNCTION admin_save_app_state(jsonb) TO anon;
